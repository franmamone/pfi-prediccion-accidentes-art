\chapter{Modelo de negocio y viabilidad econ\'omica}\label{ch:negocio}

\corr{Este cap\'{\i}tulo complementa el an\'alisis de competencias del Cap\'{\i}tulo~\ref{ch:descripcion} (FODA, Porter y Triple~P) con un modelo de sustentabilidad y un an\'alisis econ\'omico-financiero en d\'olares estadounidenses.
Los n\'umeros son supuestos explicitados, coherentes con el alcance del prototipo y con la evidencia de user research; no representan un contrato comercial firmado.}

\corr{Prevex distingue con claridad tres roles econ\'omicos \parencite{osterwalder2010}:}

	\centering
	\caption[Usuario, decisor y pagador de Prevex]{\corr{Usuario, decisor y pagador de Prevex.} \fuentepropia}
	\label{tab:negocio-roles}
	\small
	\begin{tabular}{@{}p{0.22\textwidth}p{0.68\textwidth}@{}}
		\hline
		\textbf{Rol} & \textbf{Descripci\'on} \\
		\hline
		Usuario & Preventores y coordinadores de la ART: priorizan visitas, preparan intervenciones con el copiloto normativo, registran actas y reciben asignaciones. \\
		Decisor & Gerencia o jefatura de prevenci\'on de la ART: aprueba la adopci\'on si el r\'edito operativo (cobertura, foco y trazabilidad) justifica el gasto. \\
		Pagador & La ART, mediante suscripci\'on B2B. Es quien organiza las visitas, asume obligaciones regulatorias y cubre costos de siniestralidad. \\
		\hline
	\end{tabular}
\end{table}

\corr{La cadena de valor se extiende m\'as all\'a del pagador, sin confundir beneficiario con cliente.
La ART gana eficiencia operativa (ordenamiento de visitas, dossier y normativa a mano, asignaci\'on asistida) y reduce exposici\'on a visitas mal planificadas y a costos de siniestros.
Los establecimientos se benefician si aplican las recomendaciones del preventor (menos ausentismo, mayor productividad).
Los trabajadores se benefician si el entorno laboral mejora y baja la probabilidad de accidentes leves, graves o fatales.
El contrato, no obstante, lo paga la ART.}

\section{\corr{Fuentes de ingresos y estructura de costos}}

\corr{La unidad de venta es la ART.
El mercado de referencia son las \textbf{22 aseguradoras registradas} publicadas por el Estado nacional para elecci\'on del empleador \parencite{srtartlistado}.
El precio de referencia es \textbf{el mismo} en los tres escenarios: USD~500 por mes (USD~6.000 por a\~no).
Ese precio se combina con una estructura de costos que ya incluye sueldos (no solo cloud).
Lo que diferencia pesimista, base y optimista es la \textbf{cartera fija} de clientes (3, 5 o 7 ART durante los cinco a\~nos).}

\corr{Los costos se clasifican as\'{\i}:}
\begin{itemize}
	\item \corr{\textbf{Inversi\'on inicial (capex):} USD~8.000 de productizaci\'on m\'{\i}nima (despliegue endurecido, aspectos legales livianos, materiales comerciales y primer onboarding). No es solo la factura de cloud ni de la API generativa, y no incluye las horas del PFI.}
	\item \corr{\textbf{Costos fijos anuales (opex fijo):} USD~15.000 en total, compuestos por infraestructura cloud, monitoreo y reentrenamiento (USD~3.000) m\'as \textbf{sueldos} de un perfil part-time de operaciones y soporte (USD~12.000). Sin esa partida el modelo subestimaba el costo real de sostener el producto.}
	\item \corr{\textbf{Costos variables por ART:} uso de API generativa y almacenamiento incremental (USD~800 por ART-a\~no).}
	\item \corr{\textbf{CAC:} costo de adquisici\'on al incorporar una ART nueva (USD~3.000). En cartera fija se concentra en el a\~no~1.}
\end{itemize}

\section{\corr{Viabilidad econ\'omico-financiera}}

\corr{El an\'alisis se realiza en USD (moneda estable), con horizonte de cinco a\~nos y tasa de descuento $k=18\%$ real.
Se reportan VAN, TIR, payback simple y descontado, y punto de equilibrio en n\'umero de ART.}

\corr{Se construyen tres escenarios con el mismo precio y distinta cartera fija (fracci\'on del mercado de~22 ART):}

\begin{itemize}
	\item \corr{\textbf{Pesimista:} tres ART fijas (~14\,\% del mercado) a USD~500/mes.}
	\item \corr{\textbf{Base:} cinco ART fijas (~23\,\% del mercado), mismo precio.}
	\item \corr{\textbf{Optimista:} siete ART fijas (~32\,\% del mercado), mismo precio.}
\end{itemize}

\corr{Interpretaci\'on: con sueldos incluidos, el punto de equilibrio ronda \textbf{2,9~ART}.
El pesimista (tres ART) no cierra: VAN de USD~-13.751 y no recupera la inversi\'on en cinco a\~nos.
En el escenario base el VAN es positivo (USD~13.687), la TIR (~55\,\%) supera a $k$ y el payback descontado ronda 2,5~a\~nos.
El optimista, con siete clientes, refuerza la lectura (VAN de USD~41.125; payback descontado de~1,5~a\~nos).
La lectura comercial es que el portafolio chico por debajo del equilibrio no alcanza cuando se pagan sueldos; con cinco ART el proyecto ya genera valor.}

\corr{Una sensibilidad de $k$ entre~15\% y~20\% no cambia el signo del VAN en base ni en optimista; el pesimista permanece negativo.
Los n\'umeros siguen siendo supuestos expl\'{\i}citos, no un contrato firmado.}
