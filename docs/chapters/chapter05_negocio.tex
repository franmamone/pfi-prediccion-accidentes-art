\chapter{Modelo de negocio y viabilidad econ\'omica}\label{ch:negocio}

\rev{Este cap\'{\i}tulo complementa el an\'alisis de competencias del Cap\'{\i}tulo~\ref{ch:descripcion} (FODA, Porter y Triple~P) con un modelo de sustentabilidad y un an\'alisis econ\'omico-financiero en d\'olares estadounidenses.
Los n\'umeros son supuestos explicitados, coherentes con el alcance del prototipo y con la evidencia de user research; no representan un contrato comercial firmado.}

	\begin{tabular}{@{}p{0.22\textwidth}p{0.68\textwidth}@{}}
		\hline
		\textbf{Rol} & \textbf{Descripci\'on} \\
		\hline
		Usuario & Preventores y coordinadores de la ART: priorizan visitas, preparan intervenciones con el copiloto normativo, registran actas y reciben asignaciones. \\
		Decisor & Gerencia o jefatura de prevenci\'on de la ART: aprueba la adopci\'on si el r\'edito operativo (cobertura, foco y trazabilidad) justifica el gasto. \\

		\hline
	\end{tabular}
\end{table}

\rev{La cadena de valor se extiende m\'as all\'a del pagador, sin confundir beneficiario con cliente.
La ART gana eficiencia operativa (ordenamiento de visitas, dossier y normativa a mano, asignaci\'on asistida) y reduce exposici\'on a visitas mal planificadas y a costos de siniestros.
Los establecimientos se benefician si aplican las recomendaciones del preventor (menos ausentismo, mayor productividad).
Los trabajadores se benefician si el entorno laboral mejora y baja la probabilidad de accidentes leves, graves o fatales.
El contrato, no obstante, lo paga la ART.}

\section{\rev{Fuentes de ingresos y estructura de costos}}

\rev{La unidad de venta es la ART.
El mercado de referencia son las \textbf{22 aseguradoras registradas} publicadas por el Estado nacional para elecci\'on del empleador \parencite{srtartlistado}.
El precio de referencia es \textbf{el mismo} en los tres escenarios: USD~500 por mes (USD~6.000 por a\~no).
Lo que diferencia pesimista, base y optimista es la \textbf{cartera fija} de clientes (1, 3 o 5 ART durante los cinco a\~nos), no la tarifa ni una rampa gradual.}

\rev{Los costos se clasifican as\'{\i}:}
\begin{itemize}
	\item \rev{\textbf{Inversi\'on inicial (capex):} USD~8.000 de productizaci\'on m\'{\i}nima (despliegue endurecido, aspectos legales livianos, materiales comerciales y primer onboarding). No es solo la factura de cloud ni de la API generativa, y no incluye las horas del PFI.}
	\item \rev{\textbf{Costos fijos anuales (opex fijo):} infraestructura cloud base, monitoreo, soporte y ciclo de reentrenamiento del modelo (USD~3.000, unos USD~250 por mes).}
	\item \rev{\textbf{Costos variables por ART:} uso de API generativa y almacenamiento incremental (USD~800 por ART-a\~no).}
	\item \rev{\textbf{CAC:} costo de adquisici\'on al incorporar una ART nueva (USD~3.000). En cartera fija se concentra en el a\~no~1.}
\end{itemize}

\section{\rev{Viabilidad econ\'omico-financiera}}

\rev{El an\'alisis se realiza en USD (moneda estable), con horizonte de cinco a\~nos y tasa de descuento $k=12\%$ real, alineada a un costo de oportunidad razonable para software B2B en d\'olares.
Se reportan VAN, TIR, payback simple y descontado, y punto de equilibrio en n\'umero de ART.
Los flujos salen de los supuestos del modelo de negocio (precio, adopci\'on, opex fijo, variable y CAC).}

\rev{Se construyen tres escenarios con el mismo precio y distinta cartera fija.
La cartera se expresa tambi\'en como fracci\'on del mercado de~22 ART:}

\begin{itemize}
	\item \rev{\textbf{Pesimista:} una ART fija los cinco a\~nos (~5\,\% del mercado) a USD~500/mes.}
	\item \rev{\textbf{Base:} tres ART fijas (~14\,\% del mercado), mismo precio.}
	\item \rev{\textbf{Optimista:} cinco ART fijas (~23\,\% del mercado), mismo precio.}
\end{itemize}

\rev{Interpretaci\'on: con precio \'unico, la viabilidad depende de cu\'antas ART permanecen contratadas.
En el escenario base el VAN es positivo (USD~29.384), la TIR supera a $k$ (~97\,\%) y el payback descontado ronda 1,5~a\~nos.
El optimista, con cinco clientes, refuerza esa lectura (VAN de USD~61.517; payback de~1,0~a\~no).
El pesimista, con una sola ART, queda casi neutro / levemente negativo: VAN de USD~-2.748, TIR cercana a~0\,\% y el payback no se completa en cinco a\~nos.
El ingreso anual es USD~6.000 frente a opex fijo de USD~3.000, variable de USD~800, CAC de USD~3.000 en el a\~no~1 y capex de productizaci\'on de USD~8.000.
Tras el primer a\~no el margen operativo ronda USD~2.200 anuales: insuficiente para recuperar la inversi\'on al~12\,\% en el horizonte.
La lectura comercial es que el piloto \'unico no alcanza; con tres ART el proyecto ya genera valor. El plan prioriza validar el primer contrato y cerrar un peque\~no portafolio antes de escalar CAC.}

\rev{Una sensibilidad de $k$ entre~10\% y~15\% no cambia el signo del VAN en base ni en optimista; el pesimista permanece en torno a neutro / levemente negativo.
Los n\'umeros siguen siendo supuestos expl\'{\i}citos, no un contrato firmado.}

\rev{El nombre \emph{Prevex} combina prevenci\'on con explicabilidad: el producto no solo predice, sino que expone los factores que ordenan la lista (SHAP) y apoya la visita con evidencia normativa.}

\rev{El isotipo (escudo) evoca protecci\'on laboral; la tipograf\'{\i}a del wordmark busca legibilidad en barra superior y favicon.
La paleta se mantiene sobria para un producto B2B de aseguradoras: contraste alto en UI, sin ornamentaci\'on que compita con el ranking y los estados de visita.
La marca aparece en login, encabezado y materiales de la aplicaci\'on; en este documento se presenta como identidad del prototipo evaluado, no como campa\~na comercial.}
